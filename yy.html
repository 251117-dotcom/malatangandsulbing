    <!-- Kakao Map API (services 라이브러리 포함) -->

    <script type="text/javascript" src="//dapi.kakao.com/v2/maps/sdk.js?appkey=a37189a9faf51c8c9c2b9678257633c4&libraries=services"></script>

    <script>

        // 1. 지도 생성 (기본 중심 좌표: 고양시청 근처)

        const container = document.getElementById('map');

        const options = {

            center: new kakao.maps.LatLng(37.6584, 126.8320),

            level: 5

        };

        const map = new kakao.maps.Map(container, options);


        // 2. 장소 검색 객체 및 인포윈도우 생성

        const ps = new kakao.maps.services.Places();

        const infowindow = new kakao.maps.InfoWindow({ zIndex: 1 });


        // 3. 고양시 약국 키워드 검색 실행

        ps.keywordSearch('고양시 약국', placesSearchCB);


        // 장소검색 완료 시 호출되는 콜백함수

        function placesSearchCB(data, status, pagination) {

            if (status === kakao.maps.services.Status.OK) {

                const bounds = new kakao.maps.LatLngBounds();


                for (let i = 0; i < data.length; i++) {

                    displayMarker(data[i]);

                    bounds.extend(new kakao.maps.LatLng(data[i].y, data[i].x));

                }


                // 검색된 장소 위치를 기준으로 지도 범위 재설정

                map.setBounds(bounds);

            } else if (status === kakao.maps.services.Status.ZERO_RESULT) {

                alert('검색 결과가 존재하지 않습니다.');

            } else if (status === kakao.maps.services.Status.ERROR) {

                alert('검색 결과 중 오류가 발생했습니다.');

            }

        }


        // 지도에 마커를 표시하는 함수

        function displayMarker(place) {

            const marker = new kakao.maps.Marker({

                map: map,

                position: new kakao.maps.LatLng(place.y, place.x)

            });


            // 마커 클릭 이벤트

            kakao.maps.event.addListener(marker, 'click', function() {

                const address = place.road_address_name ? place.road_address_name : place.address_name;

                const phone = place.phone ? place.phone : '전화번호 정보 없음';


                const content = `

                    <div class="info-window">

                        <div class="info-title">${place.place_name}</div>

                        <div class="info-address">${address}</div>

                        <div class="info-phone">☎ ${phone}</div>

                    </div>

                `;


                infowindow.setContent(content);

                infowindow.open(map, marker);

            });

        }
